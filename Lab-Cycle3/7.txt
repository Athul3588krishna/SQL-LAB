CREATE TABLE employee(
emp_id INT,
emp_name VARCHAR(50),
salary INT
);



CREATE TABLE Emp_history(
emp_id INT,
emp_name VARCHAR(50),
salary INT,
Date_of_deletion DATE
);



DELIMITER //

CREATE TRIGGER emp_delete_trigger
AFTER DELETE ON employee
FOR EACH ROW
BEGIN

INSERT INTO Emp_history
VALUES(OLD.emp_id, OLD.emp_name, OLD.salary, CURDATE());

END //

DELIMITER ;

CREATE DATABASE company;

USE company;

CREATE TABLE emp_details(
    EMP_ID INT PRIMARY KEY,
    FIRST_NAME VARCHAR(30),
    LAST_NAME VARCHAR(30),
    JOB_ID VARCHAR(30)
);

DELIMITER //

CREATE TRIGGER trg_emp_details
BEFORE INSERT
ON emp_details
FOR EACH ROW
BEGIN

SET NEW.FIRST_NAME = TRIM(NEW.FIRST_NAME);

SET NEW.LAST_NAME = TRIM(NEW.LAST_NAME);

SET NEW.JOB_ID = UPPER(NEW.JOB_ID);

END //

DELIMITER ;

CREATE DATABASE school;

USE school;

CREATE TABLE student_marks(
    STUDENT_ID INT PRIMARY KEY,
    NAME VARCHAR(30),
    SUB1 INT,
    SUB2 INT,
    SUB3 INT,
    SUB4 INT,
    SUB5 INT,
    TOTAL INT,
    PER_MARKS DECIMAL(5,2),
    GRADE VARCHAR(30)
);

DELIMITER //

CREATE TRIGGER trg_student_marks
AFTER UPDATE
ON student_marks
FOR EACH ROW
BEGIN

UPDATE student_marks
SET
TOTAL = NEW.SUB1 + NEW.SUB2 + NEW.SUB3 + NEW.SUB4 + NEW.SUB5,

PER_MARKS = (NEW.SUB1 + NEW.SUB2 + NEW.SUB3 + NEW.SUB4 + NEW.SUB5)/5,

GRADE =
CASE

WHEN ((NEW.SUB1 + NEW.SUB2 + NEW.SUB3 + NEW.SUB4 + NEW.SUB5)/5) >= 90
THEN 'EXCELLENT'

WHEN ((NEW.SUB1 + NEW.SUB2 + NEW.SUB3 + NEW.SUB4 + NEW.SUB5)/5) >= 75
THEN 'VERY GOOD'

WHEN ((NEW.SUB1 + NEW.SUB2 + NEW.SUB3 + NEW.SUB4 + NEW.SUB5)/5) >= 60
THEN 'GOOD'

WHEN ((NEW.SUB1 + NEW.SUB2 + NEW.SUB3 + NEW.SUB4 + NEW.SUB5)/5) >= 40
THEN 'AVERAGE'

ELSE 'NOT PROMOTED'

END

WHERE STUDENT_ID = NEW.STUDENT_ID;

END //

DELIMITER ;

UPDATE student_marks
SET
SUB1=95,
SUB2=90,
SUB3=85,
SUB4=80,
SUB5=75
WHERE STUDENT_ID=1;
