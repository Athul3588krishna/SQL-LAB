import plotly.express as px

df = px.data.gapminder().query("continent == 'Oceania'")
fig = px.line(
    df,
    x="year",
    y="lifeExp",
    color="country",
    markers=True,
    title="Life Expectancy Trends in Oceania"
)

fig.show()





import plotly.express as px

    "fruit": ["Apples", "Oranges", "Bananas", "Apples", "Oranges", "Bananas"],
    "Amount": [4, 2, 2, 4, 5, 3],
    "City": ["SF", "SF", "SF", "Montreal", "Montreal", "Montreal"]
}

fig = px.bar(
    data,
    x="fruit",
    y="Amount",
    color="City",
    barmode="group",
    title="Fruit Consumption by City"
)

fig.update_layout(template="plotly_dark")

fig.show()

4. Interactive Histogram

import plotly.express as px

# Load restaurant tipping data
df = px.data.tips()

# Create a histogram with marginal rug plots
fig = px.histogram(
    df,
    x="total_bill",
    color="sex",
    nbins=30,
    marginal="rug",   # Adds a distribution rug at the bottom
    title="Distribution of Total Bills by Gender"
)

fig.show()
